cmake_minimum_required(VERSION 3.16)
project(luka_audio)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(jsoncpp REQUIRED)
set(SENSE ${CMAKE_CURRENT_SOURCE_DIR}/../../common/vendor/sensevoice_sdk)
set(TTS ${CMAKE_CURRENT_SOURCE_DIR}/../../common/vendor/wetts_sdk)
add_executable(luka_rdk_asr native/rdk_asr.cpp)
target_include_directories(luka_rdk_asr PRIVATE ${SENSE}/include /usr/include/jsoncpp)
target_link_libraries(luka_rdk_asr ${SENSE}/lib/libsense-voice-core.a ${SENSE}/lib/libcommon.a
  ${SENSE}/lib/libggml.so ${SENSE}/lib/libggml-cpu.so ${SENSE}/lib/libggml-base.so jsoncpp pthread)
add_executable(luka_rdk_tts native/rdk_tts.cpp)
target_include_directories(luka_rdk_tts PRIVATE ${TTS}/include /usr/include/jsoncpp)
target_link_libraries(luka_rdk_tts ${TTS}/lib/libtts.so jsoncpp)
set_target_properties(luka_rdk_asr luka_rdk_tts PROPERTIES INSTALL_RPATH "$ORIGIN/rdk_lib")
install(TARGETS luka_rdk_asr luka_rdk_tts DESTINATION lib/${PROJECT_NAME})
install(FILES ${SENSE}/lib/libggml.so ${SENSE}/lib/libggml-base.so ${SENSE}/lib/libggml-cpu.so
  ${TTS}/lib/libtts.so ${TTS}/lib/libonnxruntime.so.1.11.1 DESTINATION lib/${PROJECT_NAME}/rdk_lib)
set(AIUI_SDK ${CMAKE_CURRENT_SOURCE_DIR}/vendor/aiui)
if(EXISTS ${AIUI_SDK}/libs/arm64/libaiui.so)
  add_executable(luka_aiui native/aiui.cpp)
  target_include_directories(luka_aiui PRIVATE ${AIUI_SDK}/include/aiui /usr/include/jsoncpp)
  target_link_libraries(luka_aiui ${AIUI_SDK}/libs/arm64/libaiui.so jsoncpp pthread)
  set_target_properties(luka_aiui PROPERTIES INSTALL_RPATH "$ORIGIN/aiui_lib")
  install(TARGETS luka_aiui DESTINATION lib/${PROJECT_NAME})
  install(FILES ${AIUI_SDK}/libs/arm64/libaiui.so DESTINATION lib/${PROJECT_NAME}/aiui_lib)
endif()
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/gateway DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test_aiui_error_policy test/test_aiui_error_policy.cpp)
  add_test(NAME aiui_error_policy COMMAND test_aiui_error_policy)
  add_test(NAME audio_routing COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test)
  set_tests_properties(audio_routing PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}")
endif()
ament_package()
